\begin{frame}{The $2\times2$ CI problem}
  the OSS ($^1\Sigma_u^+$) is alone; the two closed shells ($^1\Sigma_g^+$) mix:
  {\large
  \[
    \mathbf H=\begin{pmatrix}E_1&K_{12}\\K_{12}&E_2\end{pmatrix},\qquad E_1=2h_{11}+J_{11}\ (\text{RHF}),\quad E_2=2h_{22}+J_{22}
  \]}
  \vskip-0.2cm
  {\large
  \begin{align*}
    \uncover<2->{\det(\mathbf H-E\mathbf 1)}&\uncover<2->{=(E_1-E)(E_2-E)-K_{12}^2=0}\\
    \uncover<3->{E^2-(E_1+E_2)E+E_1E_2-K_{12}^2}&\uncover<3->{=0}\\
    \uncover<4->{E_{\text{CI}}}&\uncover<4->{=\tfrac12\Bigl[E_1+E_2-\sqrt{(E_2-E_1)^2+4K_{12}^2}\Bigr]}
  \end{align*}}
  \onslide<5->{\centering\small also: $E(^3\Sigma_u^+)=h_{11}+h_{22}+J_{12}-K_{12}$,\quad $E(^1\Sigma_u^+)=h_{11}+h_{22}+J_{12}+K_{12}$\par}
\end{frame}

\begin{frame}{Pulling the atoms apart (1/2): the integrals}
  \begin{assume}
    $R\to\infty$: $S\to0$, all two-centre overlap densities $\chi_A\chi_B\to0$; \ $U=\eri AAAA$, \ $V=\eri AABB\to1/R$
  \end{assume}
  {\large
  \begin{align*}
    \varphi_1^2&=\tfrac12(\chi_A^2+\chi_B^2+\canc<2->{2\chi_A\chi_B})\to\tfrac12(\chi_A^2+\chi_B^2),\qquad
      \uncover<3->{\varphi_2^2\to\tfrac12(\chi_A^2+\chi_B^2)}\\
    \uncover<4->{\varphi_1\varphi_2}&\uncover<4->{\to\tfrac12(\chi_A^2-\chi_B^2)}\\[2pt]
    \uncover<5->{J_{11}=J_{22}=J_{12}}&\uncover<5->{\to\tfrac14\bigl[U+V+V+U\bigr]=\tfrac12(U+V)}\\
    \uncover<6->{K_{12}}&\uncover<6->{\to\tfrac14\bigl[U-V-V+U\bigr]=\tfrac12(U-V)}
  \end{align*}}
  \onslide<7->{\centering\small and $h_{11}=h_{22}\to E_H-\tfrac1R$ \ (an electron on A still feels nucleus B)\par}
\end{frame}

\begin{frame}{Pulling the atoms apart (2/2): the energies}
  {\large
  \begin{align*}
    E_{\text{RHF}}&=2h_{11}+J_{11}+\tfrac1R=2E_H-\tfrac2R+\tfrac U2+\tfrac{V}{2}+\tfrac1R
      \uncover<2->{\overset{V=1/R}{=}2E_H+\hlr<3->{\tfrac U2}-\tfrac1{2R}}\\
    \uncover<4->{E_T}&\uncover<4->{=2h_{11}+J_{12}-K_{12}+\tfrac1R=2E_H-\tfrac2R+V+\tfrac1R\overset{V=1/R}{=}\hlg<5->{2E_H}}\\
    \uncover<6->{E_{\text{CI}}}&\uncover<6->{=E_1-K_{12}\ (E_1=E_2)=2E_H+\tfrac U2-\tfrac1{2R}-\tfrac{U-V}{2}\to\hlg<7->{2E_H}}\\
    \uncover<8->{E(^1\Sigma_u^+)}&\uncover<8->{=E_T+2K_{12}\to2E_H+U-V}
  \end{align*}}
  \onslide<9->{
  \begin{keyresult}
    RHF misses the dissociation limit by $U/2$; one extra configuration fixes it exactly.
  \end{keyresult}}
\end{frame}

\begin{frame}{What the CI wavefunction is}
  at $R\to\infty$ the eigenvector is $c_1=-c_2=\tfrac1{\sqrt2}$ \quad\cgold{(two configurations, equal weight)}
  {\large
  \begin{align*}
    \uncover<2->{\varphi_1(1)\varphi_1(2)}&\uncover<2->{=\tfrac12\bigl[\underbrace{\chi_A\chi_B+\chi_B\chi_A}_{\text{covalent}}
      +\underbrace{\chi_A\chi_A+\chi_B\chi_B}_{\text{ionic}}\bigr]}\\
    \uncover<3->{\varphi_1\varphi_1-\varphi_2\varphi_2}&\uncover<3->{=\tfrac12\bigl[(\chi_A+\chi_B)^2-(\chi_A-\chi_B)^2\bigr]}\\
    &\uncover<4->{=\chi_A(1)\chi_B(2)+\chi_B(1)\chi_A(2)}
  \end{align*}}
  \onslide<5->{\centering
  \begin{tikzpicture}
    \node[draw=noirrose,rounded corners,fill=noirpanel,inner sep=5pt,align=center,font=\small] at (0,0)
      {RHF: 50\% ionic\\ H$^+$H$^-$ at infinity costs $U/2$};
    \node[draw=noirgreen,rounded corners,fill=noirpanel,inner sep=5pt,align=center,font=\small] at (6,0)
      {CI: purely covalent\\ Heitler--London H$\cdot$ $\cdot$H};
  \end{tikzpicture}\par}
\end{frame}
