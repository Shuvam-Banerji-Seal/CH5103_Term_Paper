% ============================================================ Hund's rule, virial theorem, less screening
\section{Hund's rule}

\begin{frame}{Two explanations of the same rule}
  \begin{paperclaim}
    Slater's explanation: ``less electron--electron Coulomb repulsion in the triplet state'' \dots\ ``However, \dots\ the
    electron--electron repulsion is actually greater in the high-spin state'' --- hence the ``less screening rule'' (p.\ 442)
  \end{paperclaim}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \onslide<2->{
    \begin{plainpanel}
      \cgold{Slater's argument, read with frozen orbitals}\\[2pt]
      $\Delta V_{ee}=-2K<0$,\ $\Delta T=0$,\ $\Delta V_{ne}=0$\\
      \small\cmuted{triplet wins by lower repulsion}
    \end{plainpanel}}
    \column{0.48\textwidth}
    \onslide<3->{
    \begin{plainpanel}
      \cgold{Less screening (relaxed orbitals)}\\[2pt]
      $\Delta V_{ee}>0$,\ $\Delta V_{ne}\ll0$,\ $\Delta T>0$\\
      \small\cmuted{triplet wins by stronger nuclear attraction}
    \end{plainpanel}}
  \end{columns}
  \onslide<4->{\centering\small\cgold{Which is right? The virial theorem plus the computed sign of $\Delta V_{ee}$ decides.}\par}
  \symbolstrip{$\Delta X\equiv X(\text{triplet})-X(\text{singlet})$ \quad $T$: kinetic energy \quad $V_{ne}$: electron--nucleus attraction ($<0$) \quad
    $V_{ee}$: electron--electron repulsion ($>0$) \quad screening: one electron's charge partly cancelling the nucleus' pull on another}
\end{frame}

\begin{frame}{The virial theorem by scaling (1/2)}{assume an atom (nucleus at the origin); $\Psi$ exact, or optimised with respect to scaling}
  {\begin{align*}
    \stepno{1}\ \Psi_\eta(\mathbf r_1,\ldots,\mathbf r_N)&=\eta^{3N/2}\,\Psi(\eta\mathbf r_1,\ldots,\eta\mathbf r_N)
      &&\why{squeeze ($\eta>1$) or stretch ($\eta<1$) the whole cloud}\\
    \uncover<2->{\stepno{2}\ \braket{\Psi_\eta}{\Psi_\eta}}&\uncover<2->{=\eta^{3N}\!\int|\Psi(\eta\mathbf r)|^2\dd^{3N}r}
      &&\uncover<2->{\why{the prefactor appears squared}}\\
    &\uncover<3->{=\int|\Psi(\mathbf s)|^2\dd^{3N}s=1} &&\uncover<3->{\why{substitute $\mathbf s=\eta\mathbf r$, $\dd^{3N}r=\eta^{-3N}\dd^{3N}s$}}
  \end{align*}}
  \explain<4->{the scaled function is still normalised, so it is a legal trial function at every $\eta$}
  \explain<4->{the factor $\eta^{3N/2}$ exists only to keep the norm 1: $3$ coordinates per electron, $N$ electrons}
  \symbolstrip{$\eta$: scaling factor ($\eta=1$ is the original $\Psi$) \quad $N$: number of electrons \quad $\dd^{3N}r$: all $3N$ position coordinates \quad
    $\mathbf s$: rescaled coordinates}
\end{frame}

\begin{frame}{The virial theorem by scaling (2/2)}
  \begin{align*}
    \stepno{3}\ \ev{T}_\eta&=\eta^2\ev T &&\why{$T$ has two derivatives; $\nabla_{\mathbf r}=\eta\nabla_{\mathbf s}$ each}\\
    \uncover<2->{\stepno{4}\ \ev{V}_\eta}&\uncover<2->{=\eta\,\ev V} &&\uncover<2->{\why{every Coulomb term is $1/\text{distance}$: $\tfrac1r=\eta\tfrac1s$}}\\
    \uncover<3->{\stepno{5}\ E(\eta)}&\uncover<3->{=\eta^2\ev T+\eta\ev V} &&\uncover<3->{\why{add them}}\\
    \uncover<4->{\stepno{6}\ \left.\frac{\dd E}{\dd\eta}\right|_{\eta=1}}&\uncover<4->{=2\ev T+\ev V=0}
      &&\uncover<4->{\why{exact state: stationary under \emph{any} variation}}
  \end{align*}
  \onslide<5->{
  \begin{keyresult}
    \centering $2\ev T+\ev V=0\quad\Longrightarrow\quad E=\ev T+\ev V=-\ev T=\tfrac12\ev V$
  \end{keyresult}}
  \symbolstrip{$\ev T,\ev V$: expectation values of kinetic and total potential energy \quad $V=V_{ne}+V_{ee}$ for an atom \quad
    virial ratio $-\ev V/\ev T=2$ \quad Lean: \texttt{scaling\_derivative} \quad molecules: same at equilibrium geometries}
\end{frame}

\begin{frame}{Frozen orbitals cannot be the mechanism}
  Apply the virial theorem to \emph{both} exact states and subtract:
  {\[
    \Delta E\equiv E_T-E_S=-\Delta\ev T=\tfrac12\bigl(\Delta\ev{V_{ne}}+\Delta\ev{V_{ee}}\bigr)
  \]}
  \begin{enumerate}
    \item<2-> frozen-orbital model: same orbitals in both states $\Rightarrow$ $\Delta\ev T=0$
    \item<3-> then the virial theorem forces $\Delta E=0$, but the same model gives $\Delta E=-2K<0$ \ \crose{contradiction}
  \end{enumerate}
  \onslide<4->{
  \begin{keyresult}
    frozen orbitals violate the virial theorem: they cannot tell \emph{which} energy component pays
  \end{keyresult}}
  \onslide<5->{\footnotesize\cmuted{Hongo et al.\ 2004 (full text): ``This assumption leaves the kinetic energy $T$ unchanged \dots\ which
    cannot be correct as each eigenstate must satisfy the virial theorem.''}}
  \symbolstrip{same orbitals $\Rightarrow$ same $\langle T\rangle$, because $T$ is a one-electron operator summed over the same occupied orbitals \quad
    exact states must obey $E=-\langle T\rangle$, so different $E$ needs different $\langle T\rangle$}
\end{frame}

\begin{frame}{Less screening, as inequalities}{assume triplet lower ($\Delta E<0$), both states virial, $\Delta V_{ee}>0$ (what the calculations show)}
  {\begin{align*}
    \stepno{1}\ \Delta T&=-\Delta E>0 &&\why{virial: $E=-T$; electrons more compressed}\\
    \uncover<2->{\stepno{2}\ \Delta V}&\uncover<2->{=2\Delta E<0} &&\uncover<2->{\why{virial: $E=\tfrac12V$}}\\
    \uncover<3->{\stepno{3}\ \Delta V_{ne}}&\uncover<3->{=2\Delta E-\Delta V_{ee}<2\Delta E<0} &&\uncover<3->{\why{$V=V_{ne}+V_{ee}$, and $\Delta V_{ee}>0$}}\\
    \uncover<4->{\stepno{4}\ |\Delta V_{ne}|}&\uncover<4->{>\Delta V_{ee}} &&\uncover<4->{\why{from step 3}}
  \end{align*}}
  \onslide<5->{\centering\small\cgold{the nucleus wins}: stronger attraction over-compensates stronger repulsion \quad\cmuted{(Lean: \texttt{less\_screening})}\par}
  \symbolstrip{every inequality follows from the two virial identities and the measured sign of $\Delta V_{ee}$; nothing else is assumed}
\end{frame}
